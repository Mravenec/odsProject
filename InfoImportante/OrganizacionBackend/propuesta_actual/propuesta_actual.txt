package "Layer: Entities (ods_common)" {

    class Proyecto {
        - id: Integer
        - usuarioId: Integer
        - nombreProyecto: String
        - objetivoId: Integer
        - descripcion: String
        - fechaInicio: LocalDate
        - fechaFin: LocalDate
        - metaGeneral: String
        - estado: EstadoProyecto
        - createdAt: LocalDateTime
        - updatedAt: LocalDateTime
    }

    class MetaProyecto {
        - id: Integer
        - proyectoId: Integer
        - metaCodigo: String
        - metaDescripcion: String
        - valorMeta: BigDecimal
        - unidadMedida: String
        - fechaLimite: LocalDate
        - createdAt: LocalDateTime
    }

    class Indicador {
        - id: Integer
        - proyectoId: Integer
        - indicadorCodigo: String
        - indicadorDescripcion: String
        - valorActual: BigDecimal
        - valorMeta: BigDecimal
        - unidadMedida: String
        - fechaMedicion: LocalDate
        - fuenteDatos: String
        - observaciones: String
        - createdAt: LocalDateTime
        - updatedAt: LocalDateTime
    }

    class MedicionHistorica {
        - id: Integer
        - indicadorId: Integer
        - valorMedido: BigDecimal
        - fechaMedicion: LocalDate
        - responsable: String
        - metodoMedicion: String
        - createdAt: LocalDateTime
    }

    class AuditoriaOds {
        - id: Integer
        - tablaAfectada: String
        - registroId: Integer
        - accion: String
        - usuarioId: Integer
        - valoresAnteriores: String
        - valoresNuevos: String
        - fechaCambio: LocalDateTime
        - ipAddress: String
    }

    enum EstadoProyecto {
        PLANIFICACION
        ACTIVO
        COMPLETADO
        CANCELADO
    }

    Proyecto --> EstadoProyecto
    MetaProyecto --> Proyecto : proyectoId
    Indicador --> Proyecto : proyectoId
    MedicionHistorica --> Indicador : indicadorId
}

package "Layer: Entities (ods_login)" {

    class Usuario {
        - id: Integer
        - username: String
        - email: String
        - passwordHash: String
        - fullName: String
        - rolId: Integer
        - isActive: Boolean
        - emailVerificado: Boolean
        - ultimoLogin: LocalDateTime
        - intentosFallidos: Integer
        - bloqueadoHasta: LocalDateTime
        - tokenRecuperacion: String
        - tokenExpira: LocalDateTime
        - createdAt: LocalDateTime
        - updatedAt: LocalDateTime
    }

    class Rol {
        - id: Integer
        - nombre: String
        - descripcion: String
        - createdAt: LocalDateTime
    }

    class Sesion {
        - id: Integer
        - usuarioId: Integer
        - tokenHash: String
        - ipAddress: String
        - userAgent: String
        - createdAt: LocalDateTime
        - expiraEn: LocalDateTime
        - revocada: Boolean
    }

    Usuario --> Rol : rolId
    Sesion --> Usuario : usuarioId
    Proyecto --> Usuario : usuarioId
}

package "Layer: Repository Base" {

    interface IOdsBaseRepository<T> {
        ' ── Proyectos ──
        + findAllProyectos() : List<Proyecto>
        + findProyectoById(id: Integer) : Optional<Proyecto>
        + findProyectosByUsuario(usuarioId: Integer) : List<Proyecto>
        + findProyectosByEstado(estado: EstadoProyecto) : List<Proyecto>
        + saveProyecto(proyecto: Proyecto) : Proyecto
        + updateProyecto(proyecto: Proyecto) : Proyecto
        + deleteProyecto(id: Integer) : void
        ' ── Indicadores ──
        + findIndicadoresByProyecto(proyectoId: Integer) : List<Indicador>
        + findIndicadorByCodigo(proyectoId: Integer, codigo: String) : Optional<Indicador>
        + findIndicadoresByCodigoPrefix(prefix: String) : List<Indicador>
        + saveIndicador(indicador: Indicador) : Indicador
        + updateIndicador(indicador: Indicador) : Indicador
        ' ── Metas ──
        + findMetasByProyecto(proyectoId: Integer) : List<MetaProyecto>
        + saveMetaProyecto(meta: MetaProyecto) : MetaProyecto
        ' ── Mediciones ──
        + findMedicionesByIndicador(indicadorId: Integer) : List<MedicionHistorica>
        + saveMedicion(medicion: MedicionHistorica) : MedicionHistorica
        ' ── Auditoría ──
        + findAuditoriaReciente(dias: Integer) : List<AuditoriaOds>
        + findAuditoriaByRegistro(tablaAfectada: String, registroId: Integer) : List<AuditoriaOds>
        ' ── Stored Procedures ──
        + spAdminDashboard() : Map<String, Object>
        + spAdminReporteProyecto(proyectoId: Integer) : Map<String, Object>
    }

    interface ILoginRepository {
        + findUsuarioById(id: Integer) : Optional<Usuario>
        + findUsuarioByUsername(username: String) : Optional<Usuario>
        + findUsuarioByEmail(email: String) : Optional<Usuario>
        + findUsuariosActivos() : List<Usuario>
        + saveUsuario(usuario: Usuario) : Usuario
        + updateUltimoLogin(usuarioId: Integer) : void
        + incrementarIntentosFallidos(usuarioId: Integer) : void
        + bloquearUsuario(usuarioId: Integer, hasta: LocalDateTime) : void
        + findSesionByToken(tokenHash: String) : Optional<Sesion>
        + saveSesion(sesion: Sesion) : Sesion
        + revocarSesion(tokenHash: String) : void
        + revocarSesionesByUsuario(usuarioId: Integer) : void
    }
}
